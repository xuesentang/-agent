\-- 会话壳:一通对话的统一身份,messages / tickets 都引用它

CREATE TABLE conversations (

&#x20; id          BIGINT UNSIGNED NOT NULL AUTO\_INCREMENT COMMENT '会话主键',

&#x20; user\_id     VARCHAR(64)     NOT NULL                COMMENT '用户标识',

&#x20; status      ENUM('进行中','已转人工','已结束') NOT NULL DEFAULT '进行中' COMMENT '处理状态',

&#x20; created\_at  DATETIME        NOT NULL DEFAULT CURRENT\_TIMESTAMP COMMENT '开启时间',

&#x20; updated\_at  DATETIME        NOT NULL DEFAULT CURRENT\_TIMESTAMP ON UPDATE CURRENT\_TIMESTAMP COMMENT '更新时间',

&#x20; PRIMARY KEY (id),

&#x20; KEY idx\_user\_id (user\_id)

) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='客服会话';



\-- 消息流水:一通会话底下挂 N 条,role 对齐 Chat Completions 协议

CREATE TABLE messages (

&#x20; id              BIGINT UNSIGNED NOT NULL AUTO\_INCREMENT COMMENT '消息主键',

&#x20; conversation\_id BIGINT UNSIGNED NOT NULL                COMMENT '所属会话',

&#x20; role            ENUM('user','assistant','tool') NOT NULL COMMENT '角色:用户/助手/工具结果',

&#x20; content         TEXT            NULL                     COMMENT '消息正文,assistant 纯工具调用时可为空',

&#x20; tool\_calls      JSON            NULL                     COMMENT 'assistant 消息带的工具调用申请单',

&#x20; tool\_call\_id    VARCHAR(64)     NULL                     COMMENT 'tool 消息对应的申请单 id,回灌时对号入座',

&#x20; created\_at      DATETIME        NOT NULL DEFAULT CURRENT\_TIMESTAMP COMMENT '产生时间',

&#x20; PRIMARY KEY (id),

&#x20; KEY idx\_conversation\_id (conversation\_id),

&#x20; CONSTRAINT fk\_messages\_conversation FOREIGN KEY (conversation\_id) REFERENCES conversations (id)

) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='会话消息流水';



\-- FAQ 问答对:query\_faq 的数据源;ch03 起检索改走向量库,这张表退居原始录入

CREATE TABLE faq (

&#x20; id          BIGINT UNSIGNED NOT NULL AUTO\_INCREMENT COMMENT 'FAQ 主键',

&#x20; answer      TEXT            NOT NULL                COMMENT '答案',

&#x20; category    VARCHAR(64)     NOT NULL                COMMENT '分类',

&#x20; created\_at  DATETIME        NOT NULL DEFAULT CURRENT\_TIMESTAMP COMMENT '创建时间',

&#x20; updated\_at  DATETIME        NOT NULL DEFAULT CURRENT\_TIMESTAMP ON UPDATE CURRENT\_TIMESTAMP COMMENT '更新时间',

&#x20; PRIMARY KEY (id),

&#x20; KEY idx\_category (category)

) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='常见问答';



\-- 人工工单:create\_ticket 落地,工单号当业务主键

CREATE TABLE tickets (

&#x20; ticket\_no       VARCHAR(32)     NOT NULL                COMMENT '工单号,如 T20260701008',

&#x20; conversation\_id BIGINT UNSIGNED NOT NULL                COMMENT '关联会话,可倒查当时聊了什么',

&#x20; description     TEXT            NOT NULL                COMMENT '问题描述',

&#x20; ticket\_type     ENUM('售后','投诉','咨询') NOT NULL     COMMENT '工单类型',

&#x20; status          ENUM('待处理','已处理') NOT NULL DEFAULT '待处理' COMMENT '处理状态',

&#x20; created\_at      DATETIME        NOT NULL DEFAULT CURRENT\_TIMESTAMP COMMENT '创建时间',

&#x20; PRIMARY KEY (ticket\_no),

&#x20; KEY idx\_conversation\_id (conversation\_id),

\-- ch04 编造个案台账:忠实度裁判判出的每条编造,不只留在这一轮的报告里,进表长期管理。

\--

\-- 为什么要一张表:报告是产物,重跑一次就被覆盖,上一轮判出的个案连带它的处置状态一起没了。

\-- 而这些个案的价值恰恰在跨轮追溯——同一道题反复被判编造,说明库里那一格一直没补对。

\--

\-- 一题一行(uk\_eval\_id):同一道题再次被判编造不新增行,只把答案/理由/角标快照更新成最近一次、

\-- seen\_count 加一。已经标「已解决」的题又被判出来,状态自动退回「未解决」——那是复发,

\-- 不是新问题,得让它重新出现在待处理列表里。

\--

\-- citations 存的是这一轮喂给模型的 Top-K 证据\*\*全集\*\*(答案里的角标 \[n] 就是这份列表的序号)。

\-- 裁判说「证据里没有」,追溯时必须能当场看到当时喂进去的到底是什么,不能只留一句结论;

\-- 而且要看得出「手里有哪几条、实际只引了哪几条」——没被引用的那些同样是判断依据。

CREATE TABLE faith\_cases (

&#x20; id            BIGINT UNSIGNED NOT NULL AUTO\_INCREMENT COMMENT '主键',

&#x20; eval\_id       VARCHAR(16)     NOT NULL                COMMENT '评估集题号,如 A43;一题一行',

&#x20; bucket        VARCHAR(24)     NOT NULL                COMMENT '题目所属桶:A\_policy / B\_model / C\_colloquial / E\_multi',

&#x20; query         VARCHAR(512)    NOT NULL                COMMENT '用户问题原文',

&#x20; strategy      VARCHAR(24)     NOT NULL DEFAULT 'hybrid\_rerank' COMMENT '产出这条答案的检索策略',

&#x20; answer        TEXT            NOT NULL                COMMENT '被判编造的那版生成答案原文',

&#x20; reason        TEXT            NOT NULL                COMMENT '裁判给的理由:编在哪一句',

&#x20; citations     JSON            NULL                    COMMENT '这一轮喂给模型的 Top-K 证据全集快照:\[{n,chunk\_id,section\_path,question,answer}];答案里的角标 \[n] 就是这份列表的序号,答案通常只引用其中两三条;老数据没记为 NULL',

&#x20; judge\_model   VARCHAR(64)     NULL                    COMMENT '判这条的裁判模型',

&#x20; status        ENUM('未解决','已解决','无需解决') NOT NULL DEFAULT '未解决' COMMENT '处置状态,人工点按钮改',

&#x20; seen\_count    INT UNSIGNED    NOT NULL DEFAULT 1      COMMENT '被判编造的累计次数(跨轮)',

&#x20; first\_seen\_at DATETIME        NOT NULL DEFAULT CURRENT\_TIMESTAMP COMMENT '第一次被判编造的时间',

&#x20; last\_seen\_at  DATETIME        NOT NULL DEFAULT CURRENT\_TIMESTAMP COMMENT '最近一次被判编造的时间',

&#x20; resolution    VARCHAR(300)    NULL                    COMMENT '处置说明:标已解决要写清怎么解决的,标无需解决要写清为什么不用改;退回未解决时清空。空着的处置在台账上等于没有交代',

&#x20; resolved\_at   DATETIME        NULL                    COMMENT '最近一次被标为已解决/无需解决的时间;复发后仍保留,用来标「复发」',

&#x20; PRIMARY KEY (id),

&#x20; UNIQUE KEY uk\_eval\_id (eval\_id),

&#x20; KEY idx\_status (status),

&#x20; KEY idx\_last\_seen\_at (last\_seen\_at)

) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='ch04 忠实度编造个案台账';
